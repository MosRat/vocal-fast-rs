cmake_minimum_required(VERSION 3.24)
project(vocal_cpu_operator_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(dnnl CONFIG REQUIRED)
option(VOCAL_BENCH_LEGACY_PROVIDERS "Build ATen/XNNPACK migration comparison providers" ON)
function(add_provider provider)
  add_library(bench_${provider} SHARED common.cpp ${provider}.cpp)
  target_compile_definitions(bench_${provider} PRIVATE NOMINMAX)
  target_compile_options(bench_${provider} PRIVATE /O2 /EHsc /Zc:__cplusplus)
endfunction()
add_provider(onednn)
target_link_libraries(bench_onednn PRIVATE DNNL::dnnl)
add_library(bench_onednn_int8 SHARED common.cpp onednn.cpp)
target_compile_definitions(bench_onednn_int8 PRIVATE NOMINMAX VOCAL_BENCH_INT8)
target_compile_options(bench_onednn_int8 PRIVATE /O2 /EHsc /Zc:__cplusplus)
target_link_libraries(bench_onednn_int8 PRIVATE DNNL::dnnl)
if(VOCAL_BENCH_LEGACY_PROVIDERS)
  if(NOT EXISTS "${TORCH_SDK}/include/xnnpack.h")
    message(FATAL_ERROR "TORCH_SDK required for legacy providers; use -DVOCAL_BENCH_LEGACY_PROVIDERS=OFF for standalone oneDNN/INT8")
  endif()
  add_provider(aten)
  add_provider(xnnpack)
  target_include_directories(bench_aten PRIVATE "${TORCH_SDK}/include" "${TORCH_SDK}/include/torch/csrc/api/include")
  target_link_libraries(bench_aten PRIVATE "${TORCH_SDK}/lib/torch_cpu.lib" "${TORCH_SDK}/lib/c10.lib")
  target_include_directories(bench_xnnpack PRIVATE "${TORCH_SDK}/include")
  target_sources(bench_xnnpack PRIVATE layout.cpp)
  set_source_files_properties(layout.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  # Standalone archives; no torch/c10 tensor runtime is linked to XNNPACK.
  foreach(lib IN ITEMS XNNPACK microkernels-prod pthreadpool cpuinfo)
    target_link_libraries(bench_xnnpack PRIVATE "${TORCH_SDK}/lib/${lib}.lib")
  endforeach()
endif()
